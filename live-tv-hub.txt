// ── BUILD DL STREAMS 1–502 ──
(function() {
  var container = document.getElementById('dl-list');
  var frag = document.createDocumentFragment();
  for (var n = 1; n <= 502; n++) {
    var num = (n < 10) ? '0' + n : '' + n;
    var d = document.createElement('div');
    d.className = 'ch';
    d.setAttribute('data-url', 'https://dlstreams.com/stream/stream-' + num + '.php');
    d.setAttribute('data-name', 'Stream ' + n);
    d.textContent = 'Stream ' + n;
    frag.appendChild(d);
  }
  container.appendChild(frag);
})();

// ── INIT ──
var channels    = document.querySelectorAll('.ch');
var player      = document.getElementById('player');
var nowEl       = document.getElementById('now');
var loader      = document.getElementById('loader');
var loadName    = document.getElementById('loading-name');
var sidebar     = document.getElementById('sidebar');
var focusIdx    = 0;
var loaderTimer = null;

// ── PLAY ──
function play(el) {
  for (var i = 0; i < channels.length; i++) channels[i].classList.remove('on');
  el.classList.add('on');
  var url  = el.getAttribute('data-url');
  var name = el.getAttribute('data-name');
  loadName.textContent = name;
  loader.classList.add('show');
  clearTimeout(loaderTimer);
  loaderTimer = setTimeout(function() {
    player.src = url;
    nowEl.textContent = name;
  }, 300);
  loaderTimer = setTimeout(function() {
    loader.classList.remove('show');
  }, 5000);
}

player.addEventListener('load', function() {
  clearTimeout(loaderTimer);
  setTimeout(function(){ loader.classList.remove('show'); }, 500);
});

// ── CLICK ──
for (var i = 0; i < channels.length; i++) {
  (function(idx, el) {
    el.addEventListener('click', function() {
      focusIdx = idx;
      setFocus(idx);
      play(el);
    });
  })(i, channels[i]);
}

// ── FOCUS / REMOTE ──
function setFocus(idx) {
  for (var i = 0; i < channels.length; i++) channels[i].classList.remove('focused');
  if (idx >= 0 && idx < channels.length) {
    channels[idx].classList.add('focused');
    channels[idx].scrollIntoView({ block: 'nearest' });
  }
}

sidebar.focus();
setFocus(0);

document.addEventListener('keydown', function(e) {
  var key = e.keyCode;
  if (key === 40 || key === 34) {          // Down / PageDown
    e.preventDefault();
    if (focusIdx < channels.length - 1) { focusIdx++; setFocus(focusIdx); }
  } else if (key === 38 || key === 33) {   // Up / PageUp
    e.preventDefault();
    if (focusIdx > 0) { focusIdx--; setFocus(focusIdx); }
  } else if (key === 13 || key === 32) {   // Enter / OK
    e.preventDefault();
    play(channels[focusIdx]);
  } else if (key >= 49 && key <= 57) {     // 1–9 quick jump
    var n = key - 49;
    if (n < channels.length) { focusIdx = n; setFocus(n); play(channels[n]); }
  }
});
